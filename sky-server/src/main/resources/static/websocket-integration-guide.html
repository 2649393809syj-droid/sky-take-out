步骤2：添加 WebSocket 连接代码

方案A：Vue 项目（推荐）

在 src/main.js 中添加：

import Vue from 'vue'
import ElementUI from 'element-ui'

Vue.use(ElementUI)

// WebSocket 全局配置
Vue.prototype.$websocket = null

Vue.prototype.$connectWebSocket = function() {
    if (this.$websocket && this.$websocket.readyState === WebSocket.OPEN) {
        return
    }
    
    // 动态获取 WebSocket 地址
    const wsProtocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:'
    const wsHost = window.location.host
    const wsUrl = `${wsProtocol}//${wsHost}/ws/admin`
    
    this.$websocket = new WebSocket(wsUrl)
    
    this.$websocket.onopen = () => {
        console.log("WebSocket连接成功")
    }
    
    this.$websocket.onmessage = (event) => {
        console.log("收到消息：", event.data)
        const data = JSON.parse(event.data)
        
        if (data.type === 1) {
            // 新订单通知
            this.$notify({
                title: '新订单提醒',
                message: data.content,
                type: 'success',
                duration: 0,
                position: 'top-right'
            })
            
            // 播放提示音
            const audio = new Audio('/audio/preview.mp3')
            audio.play()
            
            // 语音播报
            if ('speechSynthesis' in window) {
                const utterance = new SpeechSynthesisUtterance("您有新的订单，请及时处理")
                utterance.lang = 'zh-CN'
                window.speechSynthesis.speak(utterance)
            }
        } else if (data.type === 2) {
            // 客户催单
            this.$notify({
                title: '客户催单',
                message: data.content,
                type: 'warning',
                duration: 0,
                position: 'top-right'
            })
            
            const audio = new Audio('/audio/reminder.mp3')
            audio.play()
        }
    }
    
    this.$websocket.onclose = () => {
        console.log("WebSocket连接关闭，5秒后重连")
        setTimeout(() => this.$connectWebSocket(), 5000)
    }
    
    this.$websocket.onerror = (error) => {
        console.error("WebSocket错误", error)
    }
}

// 在 Vue 实例创建时连接
new Vue({
    router,
    store,
    render: h => h(App),
    mounted() {
        // 连接 WebSocket
        this.$connectWebSocket()
        
        // 请求通知权限
        if ('Notification' in window && Notification.permission === 'default') {
            Notification.requestPermission()
        }
    }
}).$mount('#app')

方案B：在任意页面中使用（通用方案）

在你的主页面 HTML 的 <script> 标签中添加：

<script>
// 全局 WebSocket 对象
let websocket = null;

// 连接 WebSocket
function connectWebSocket() {
    if (websocket && websocket.readyState === WebSocket.OPEN) {
        return;
    }
    
    // 动态获取 WebSocket 地址
    const wsProtocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
    const wsHost = window.location.host;
    const wsUrl = `${wsProtocol}//${wsHost}/ws/admin`;
    
    websocket = new WebSocket(wsUrl);
    
    websocket.onopen = function() {
        console.log("WebSocket连接成功");
    };
    
    websocket.onmessage = function(event) {
        console.log("收到消息：", event.data);
        
        try {
            const data = JSON.parse(event.data);
            
            if (data.type === 1) {
                // 新订单通知
                showNotification('新订单提醒', data.content);
                playSound('/audio/preview.mp3');
                playVoice('您有新的订单，请及时处理');
            } else if (data.type === 2) {
                // 客户催单
                showNotification('客户催单', data.content);
                playSound('/audio/reminder.mp3');
                playVoice('有客户催单，请尽快处理');
            }
        } catch (e) {
            console.error("解析消息失败", e);
        }
    };
    
    websocket.onclose = function() {
        console.log("WebSocket连接关闭，5秒后重连");
        setTimeout(connectWebSocket, 5000);
    };
    
    websocket.onerror = function(error) {
        console.error("WebSocket错误", error);
    };
}

// 显示通知（使用 Element UI）
function showNotification(title, message) {
    if (typeof ElementUI !== 'undefined') {
        ElementUI.Notification({
            title: title,
            message: message,
            type: title.includes('催单') ? 'warning' : 'success',
            duration: 0,
            position: 'top-right'
        });
    } else {
        // 使用浏览器原生通知
        if (Notification.permission === "granted") {
            new Notification(title, { body: message });
        }
    }
}

// 播放提示音
function playSound(audioUrl) {
    const audio = new Audio(audioUrl);
    audio.play().catch(e => console.log('播放提示音失败', e));
}

// 语音播报
function playVoice(text) {
    if ('speechSynthesis' in window) {
        window.speechSynthesis.cancel();
        const utterance = new SpeechSynthesisUtterance(text);
        utterance.lang = 'zh-CN';
        utterance.rate = 1.0;
        utterance.pitch = 1.0;
        window.speechSynthesis.speak(utterance);
    }
}

// 页面加载时初始化
window.addEventListener('load', function() {
    // 请求通知权限
    if ('Notification' in window && Notification.permission === 'default') {
        Notification.requestPermission();
    }
    
    // 连接 WebSocket
    connectWebSocket();
});

// 页面卸载时关闭连接
window.addEventListener('beforeunload', function() {
    if (websocket) {
        websocket.close();
    }
});
</script>

常见问题

Q: 找不到前端项目在哪里？

A: 你的前端项目应该是一个独立的文件夹，通常名为 sky-admin、admin-web 或类似的名字。如果你是从课程资料中获取的，检查一下是否有单独的前端压缩包。

Q: 如何知道我用的是什么前端框架？

A: 查看前端项目的 package.json 文件，在 dependencies 中可以看到：

  • 如果有 "vue"，就是 Vue 项目
  • 如果有 "react"，就是 React 项目
  • 如果有 "element-ui"，说明使用了 Element UI 组件库

Q: 代码应该加在文件的什么位置？

A:

  • Vue 项目：在 main.js 的最后，new Vue({...}).$mount('#app') 这部分
  • HTML 文件：在 </body> 标签之前添加 <script> 标签
